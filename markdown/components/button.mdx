# 버튼 (Button)

어떤 환경(React, Vue, Angular, Vanilla JS, jQuery)에서든 즉시 동작하는 범용 인터랙티브 버튼입니다. 디자인 토큰을 통해 모든 테마와 브랜드 스타일이 자동 반영됩니다.

<Callout type="tip">
  버튼 컴포넌트는 표준 HTML 속성을 100% 보존하며, W3C 웹 접근성 표준(A11y)과 키보드 인터랙션을 완벽히 지원합니다.
</Callout>

## 🚀 멀티 프레임워크 사용법 (Multi-Framework Usage)

<FrameworkTabs 
  react={`import 'atomsystem/css/atom-ui.css';
import { Button } from 'atomsystem';

export default function Demo() {
  return (
    <div className="flex gap-3">
      <Button type="primary">확인</Button>
      <Button type="secondary">취소</Button>
      <Button disabled>비활성화</Button>
    </div>
  );
}`}
  vue={`<script setup>
// atomsystem/css/atom-ui.css 전역 로드
</script>

<template>
  <div class="flex gap-3">
    <button class="atom-btn atom-btn-primary" @click="handleConfirm">확인</button>
    <button class="atom-btn atom-btn-secondary" @click="handleCancel">취소</button>
    <button class="atom-btn" disabled>비활성화</button>
  </div>
</template>`}
  vanilla={`<!-- 1. CSS 로드 -->
<link rel="stylesheet" href="node_modules/atomsystem/css/atom-ui.css">

<!-- 2. 순수 HTML 마크업 -->
<button class="atom-btn atom-btn-primary" id="btn-submit">확인</button>
<button class="atom-btn atom-btn-secondary" id="btn-cancel">취소</button>
<button class="atom-btn" disabled>비활성화</button>

<!-- jQuery 예시 -->
<script>
  $('#btn-submit').on('click', function() {
    alert('확인 클릭됨!');
  });
</script>`}
  angular={`import { Component } from '@angular/core';

@Component({
  selector: 'app-button-demo',
  standalone: true,
  template: \`
    <div class="flex gap-3">
      <button class="atom-btn atom-btn-primary" (click)="onConfirm()">확인</button>
      <button class="atom-btn atom-btn-secondary" (click)="onCancel()">취소</button>
    </div>
  \`
})
export class ButtonDemoComponent {
  onConfirm() { console.log('Confirm'); }
  onCancel() { console.log('Cancel'); }
}`}
/>
